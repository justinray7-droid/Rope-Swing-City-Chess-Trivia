<script>
const cvs=document.getElementById('c');
const ctx=cvs.getContext('2d');
let W,H;
function resize(){W=cvs.width=window.innerWidth;H=cvs.height=window.innerHeight;}
resize();
window.addEventListener('resize',resize);

const oEl=document.getElementById('overlay');
const oTitle=document.getElementById('oTitle');
const oMsg=document.getElementById('oMsg');
const oBtn=document.getElementById('oBtn');
const scoreEl=document.getElementById('scoreVal');
const levelEl=document.getElementById('levelVal');
const livEl=document.getElementById('livVal');
const hintEl=document.getElementById('hint');
const timerBox=document.getElementById('timerBox');

const GRAVITY=0.18;
const PR=14;
const CTYPES=[
  {label:'coin', c:'#ffd700',rc:'#e65100',pts:10,r:9},
  {label:'star', c:'#fff176',rc:'#f9a825',pts:25,r:11},
  {label:'gem',  c:'#80deea',rc:'#0097a7',pts:50,r:9},
];
const BCOLORS=[
  ['#b71c1c','#e53935'],['#1565c0','#1e88e5'],['#2e7d32','#43a047'],
  ['#6a1b9a','#8e24aa'],['#e65100','#fb8c00'],['#00695c','#00897b'],
  ['#37474f','#546e7a'],['#4a148c','#7b1fa2'],
];
const SKIES=[
  ['#0d1b2a','#1b2838','#2d3f55'],
  ['#0a0a25','#131340','#1e1e5e'],
  ['#050510','#0c0c30','#141450'],
  ['#050508','#080820','#101038'],
  ['#030305','#060615','#0a0a28'],
];

let game=null;
let currentLevel=1;
let pressStart=0;

let timerStart=null;
let timerRunning=false;
function tickTimer(){
  if(timerRunning&&timerStart!==null){
    const el=Math.floor((Date.now()-timerStart)/1000);
    const m=String(Math.floor(el/60)).padStart(2,'0');
    const s=String(el%60).padStart(2,'0');
    timerBox.textContent='\u23f1 '+m+':'+s;
  }
}
setInterval(tickTimer,250);

function rrect(x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y);
  ctx.lineTo(x+w-r,y);ctx.arcTo(x+w,y,x+w,y+r,r);
  ctx.lineTo(x+w,y+h-r);ctx.arcTo(x+w,y+h,x+w-r,y+h,r);
  ctx.lineTo(x+r,y+h);ctx.arcTo(x,y+h,x,y+h-r,r);
  ctx.lineTo(x,y+r);ctx.arcTo(x,y,x+r,y,r);
  ctx.closePath();
}

class Game{
  constructor(lvl){
    this.lvl=lvl;this.score=0;this.lives=3;this.state='playing';
    this.camX=0;this.particles=[];this.floats=[];
    this.px=80;this.py=H*0.5;this.pvx=0;this.pvy=0;
    this.grounded=false;this.swinging=false;this.throwing=false;
    this.anchor=null;this.ropeLen=0;this.swingAngle=0;this.swingOmega=0;
    this.ropeHead=null;this.ropeVel=null;this.ropeTarget=null;
    this.groundY=H-32;this.gen();
  }
  gen(){
    const lvl=this.lvl;
    this.buildings=[];this.collectibles=[];
    const bCount=5+lvl*2;
    let x=160;
    for(let i=0;i<bCount;i++){
      const bw=75+Math.random()*90;
      const bh=260+Math.random()*320;
      const by=this.groundY-bh;
      const ci=i%BCOLORS.length;
      const b={x,y:by,w:bw,h:bh,c1:BCOLORS[ci][0],c2:BCOLORS[ci][1],wins:[]};
      const cols=Math.max(2,Math.floor(bw/26));
      const rows=Math.max(3,Math.floor(bh/32));
      for(let r=1;r<=rows;r++) for(let c=1;c<=cols;c++){
        b.wins.push({x:b.x+c*(bw/(cols+1))-5, y:b.y+r*(bh/(rows+1))-5, lit:Math.random()>0.4});
      }
      this.buildings.push(b);
      const n=1+Math.floor(Math.random()*3);
      for(let j=0;j<n;j++){
        const ct=CTYPES[Math.floor(Math.random()*CTYPES.length)];
        this.collectibles.push({
          x:x+bw*0.1+Math.random()*bw*0.8,
          y:by+bh*0.25+Math.random()*bh*0.4,
          type:ct, r:ct.r, bob:Math.random()*Math.PI*2, collected:false
        });
      }
      x+=bw+55+Math.random()*80+(lvl-1)*12;
    }
    this.flagX=x+30;this.flagY=this.groundY-220;
    this.worldW=this.flagX+200;
    this.skyColors=SKIES[Math.min(lvl-1,SKIES.length-1)];
  }
  findAnchor(wx,wy){
    let best=null,bd=9999;
    for(const b of this.buildings){
      const pts=[{x:b.x,y:b.y},{x:b.x+b.w,y:b.y},{x:b.x+b.w/2,y:b.y}];
      for(const p of pts){
        if(p.y>=this.py) continue;
        const d=Math.hypot(p.x-wx,p.y-wy);
        if(d<bd&&d<520){bd=d;best=p;}
      }
    }
    return best;
  }
  throwRope(sx,sy){
    const wx=sx+this.camX, wy=sy;
    const a=this.findAnchor(wx,wy);
    if(!a) return;
    this.throwing=true;this.swinging=false;
    this.ropeTarget=a;this.ropeHead={x:this.px,y:this.py};
    const dx=a.x-this.px, dy=a.y-this.py;
    const len=Math.sqrt(dx*dx+dy*dy)||1;
    const spd=28;
    this.ropeVel={x:dx/len*spd,y:dy/len*spd};
    this.anchor=null;
  }
  attach(){
    this.throwing=false;this.swinging=true;
    this.anchor={x:this.ropeHead.x,y:this.ropeHead.y};
    const dx=this.px-this.anchor.x, dy=this.py-this.anchor.y;
    this.ropeLen=Math.sqrt(dx*dx+dy*dy)||80;
    this.swingAngle=Math.atan2(dx, dy);
    const r=this.ropeLen;
    this.pvy=-6;this.pvx+=1.9;
    let omega=(this.pvx*Math.cos(this.swingAngle)-this.pvy*Math.sin(this.swingAngle))/r;
    this.swingOmega=Math.abs(omega)+0.019;
  }
  release(){
    if(!this.swinging) return;
    const r=this.ropeLen;
    const tang=this.swingOmega*r;
    this.pvx=tang*Math.cos(this.swingAngle);
    this.pvy=-tang*Math.sin(this.swingAngle);
    this.swinging=false;this.anchor=null;
  }
  update(){
    if(this.state!=='playing') return;
    if(this.throwing&&this.ropeHead){
      this.ropeHead.x+=this.ropeVel.x;this.ropeHead.y+=this.ropeVel.y;
      const t=this.ropeTarget;
      if(Math.hypot(this.ropeHead.x-t.x,this.ropeHead.y-t.y)<18){
        this.ropeHead.x=t.x;this.ropeHead.y=t.y;this.attach();
      }
      if(this.ropeHead.y>H+80) this.throwing=false;
    }
    if(this.swinging&&this.anchor){
      const alpha=-(GRAVITY/this.ropeLen)*Math.sin(this.swingAngle);
      this.swingOmega+=alpha;this.swingOmega*=0.999;this.swingAngle+=this.swingOmega;
      this.px=this.anchor.x+this.ropeLen*Math.sin(this.swingAngle);
      this.py=this.anchor.y+this.ropeLen*Math.cos(this.swingAngle);
      this.pvx=this.swingOmega*this.ropeLen*Math.cos(this.swingAngle);
      this.pvy=-this.swingOmega*this.ropeLen*Math.sin(this.swingAngle);
    } else if(!this.swinging){
      this.pvy+=GRAVITY;this.px+=this.pvx;this.py+=this.pvy;this.pvx*=0.99;
      if(this.py>=this.groundY-PR){
        if(this.pvy>10){this.die();return;}
        this.py=this.groundY-PR;this.pvy=0;this.pvx*=0.8;this.grounded=true;
      } else {this.grounded=false;}
    }
    if(this.py>H+120){this.die();return;}
    const tx=this.px-W*0.35;
    this.camX+=(Math.max(0,Math.min(this.worldW-W,tx))-this.camX)*0.1;
    for(const c of this.collectibles){
      if(c.collected) continue;
      c.bob+=0.05;
      const cy=c.y+Math.sin(c.bob)*5;
      if(Math.hypot(this.px-c.x,this.py-cy)<c.r+PR){
        c.collected=true;this.score+=c.type.pts;scoreEl.textContent=this.score;
        this.burst(c.x,cy,c.type.c,14);
        this.floats.push({x:c.x,y:cy,txt:'+'+c.type.pts,a:1,vy:-1.6,c:c.type.c});
      }
    }
    if(this.px>=this.flagX) this.state='win';
    this.particles=this.particles.filter(p=>{
      p.x+=p.vx;p.y+=p.vy;p.vy+=0.12;p.life--;p.a=p.life/p.ml;return p.life>0;
    });
    this.floats=this.floats.filter(f=>{f.y+=f.vy;f.a-=0.014;return f.a>0;});
  }
  die(){
    this.lives=Math.max(0,this.lives-1);
    this.burst(this.px,this.py,'#ff5252',22);
    if(this.lives<=0){this.state='dead';return;}
    this.px=80;this.py=H*0.5;this.pvx=0;this.pvy=0;
    this.swinging=false;this.throwing=false;this.anchor=null;this.camX=0;
    livEl.textContent='\u2764\uFE0F'.repeat(this.lives);
  }
  burst(x,y,col,n){
    for(let i=0;i<n;i++){
      const a=Math.random()*Math.PI*2,s=2+Math.random()*5;
      this.particles.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,a:1,ml:45,life:45,r:2+Math.random()*3,c:col});
    }
  }
  draw(){
    const sk=this.skyColors;
    const gr=ctx.createLinearGradient(0,0,0,H);
    gr.addColorStop(0,sk[0]);gr.addColorStop(0.55,sk[1]);gr.addColorStop(1,sk[2]);
    ctx.fillStyle=gr;ctx.fillRect(0,0,W,H);
    for(let i=0;i<90;i++){
      const sx=(i*173.7)%W, sy=(i*97.3)%(H*0.65);
      ctx.globalAlpha=0.25+Math.sin(Date.now()*0.0008+i)*0.18;
      ctx.fillStyle='#fff';
      ctx.beginPath();ctx.arc(sx,sy,0.7,0,Math.PI*2);ctx.fill();
    }
    ctx.globalAlpha=1;
    ctx.save();ctx.translate(-this.camX,0);
    for(const b of this.buildings){
      ctx.fillStyle='rgba(0,0,0,0.25)';rrect(b.x+6,b.y+6,b.w,b.h,3);ctx.fill();
      const bg=ctx.createLinearGradient(b.x,b.y,b.x+b.w,b.y+b.h);
      bg.addColorStop(0,b.c1);bg.addColorStop(1,b.c2);
      ctx.fillStyle=bg;rrect(b.x,b.y,b.w,b.h,3);ctx.fill();
      for(const w of b.wins){
        ctx.fillStyle=w.lit?'#fff9c4':'rgba(0,0,0,0.35)';
        ctx.fillRect(w.x,w.y,11,9);
        if(w.lit){ctx.fillStyle='rgba(255,255,180,0.12)';ctx.fillRect(w.x-1,w.y-1,13,11);}
      }
    }
    const gg=ctx.createLinearGradient(0,this.groundY,0,H);
    gg.addColorStop(0,'#1b2631');gg.addColorStop(1,'#0d1117');
    ctx.fillStyle=gg;ctx.fillRect(0,this.groundY,this.worldW,H-this.groundY);
    ctx.strokeStyle='rgba(255,255,255,0.12)';ctx.lineWidth=3;ctx.setLineDash([28,18]);
    ctx.beginPath();ctx.moveTo(0,this.groundY+14);ctx.lineTo(this.worldW,this.groundY+14);ctx.stroke();
    ctx.setLineDash([]);
    for(const c of this.collectibles){
      if(c.collected) continue;
      const cy=c.y+Math.sin(c.bob)*5;
      ctx.save();ctx.translate(c.x,cy);
      ctx.shadowColor=c.type.c;ctx.shadowBlur=12;
      ctx.fillStyle=c.type.rc;ctx.beginPath();ctx.arc(0,0,c.r+3,0,Math.PI*2);ctx.fill();
      ctx.shadowBlur=0;
      ctx.fillStyle=c.type.c;ctx.beginPath();ctx.arc(0,0,c.r,0,Math.PI*2);ctx.fill();
      ctx.fillStyle='rgba(255,255,255,0.45)';ctx.beginPath();ctx.arc(-2,-2,c.r*0.33,0,Math.PI*2);ctx.fill();
      ctx.restore();
    }
    this.drawFlag();
    if(this.throwing&&this.ropeHead){
      ctx.strokeStyle='#a5d6a7';ctx.lineWidth=2;ctx.lineCap='round';
      ctx.beginPath();ctx.moveTo(this.px,this.py);ctx.lineTo(this.ropeHead.x,this.ropeHead.y);ctx.stroke();
    }
    if(this.swinging&&this.anchor){
      const mx=(this.px+this.anchor.x)/2+4, my=(this.py+this.anchor.y)/2+8;
      ctx.strokeStyle='#a5d6a7';ctx.lineWidth=2.5;ctx.lineCap='round';
      ctx.beginPath();ctx.moveTo(this.anchor.x,this.anchor.y);
      ctx.quadraticCurveTo(mx,my,this.px,this.py);ctx.stroke();
      ctx.fillStyle='#ffee58';ctx.shadowColor='#ffee58';ctx.shadowBlur=8;
      ctx.beginPath();ctx.arc(this.anchor.x,this.anchor.y,5,0,Math.PI*2);ctx.fill();
      ctx.shadowBlur=0;
    }
    this.drawPlayer();
    for(const p of this.particles){
      ctx.globalAlpha=p.a;ctx.fillStyle=p.c;
      ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,Math.PI*2);ctx.fill();
    }
    ctx.globalAlpha=1;
    ctx.textAlign='center';ctx.font='bold 17px Segoe UI';
    for(const f of this.floats){ctx.globalAlpha=f.a;ctx.fillStyle=f.c;ctx.fillText(f.txt,f.x,f.y);}
    ctx.globalAlpha=1;ctx.restore();
  }
  drawPlayer(){
    const x=this.px,y=this.py;
    ctx.fillStyle='#ff7043';ctx.beginPath();ctx.arc(x,y,PR,0,Math.PI*2);ctx.fill();
    ctx.fillStyle='#ffccbc';ctx.beginPath();ctx.arc(x,y-1,PR-4,0,Math.PI*2);ctx.fill();
    ctx.fillStyle='#37474f';
    ctx.beginPath();ctx.arc(x-4,y-4,2,0,Math.PI*2);ctx.fill();
    ctx.beginPath();ctx.arc(x+4,y-4,2,0,Math.PI*2);ctx.fill();
    ctx.strokeStyle='#37474f';ctx.lineWidth=1.5;
    ctx.beginPath();ctx.arc(x,y-1,5,0.15,Math.PI-0.15);ctx.stroke();
    ctx.fillStyle='#1565c0';ctx.beginPath();ctx.ellipse(x,y-PR+3,PR-1,8,0,Math.PI,0);ctx.fill();
    if(this.swinging){
      ctx.strokeStyle='#ef5350';ctx.lineWidth=4;ctx.lineCap='round';
      ctx.beginPath();ctx.moveTo(x,y);ctx.lineTo(x-this.pvx*2.5,y-this.pvy*2.5);ctx.stroke();
    }
  }
  drawFlag(){
    const x=this.flagX;
    const fg=ctx.createLinearGradient(x-12,0,x+12,0);
    fg.addColorStop(0,'rgba(229,57,53,0)');fg.addColorStop(0.5,'rgba(229,57,53,0.5)');fg.addColorStop(1,'rgba(229,57,53,0)');
    ctx.fillStyle=fg;ctx.fillRect(x-12,0,24,H);
    ctx.strokeStyle='#ff5252';ctx.lineWidth=3;
    ctx.setLineDash([14,10]);ctx.lineDashOffset=-Date.now()*0.04;
    ctx.beginPath();ctx.moveTo(x,0);ctx.lineTo(x,H);ctx.stroke();
    ctx.setLineDash([]);
    for(let i=0;i<2;i++) for(let j=0;j<6;j++){
      ctx.fillStyle=((i+j)%2===0)?'#fff':'#e53935';
      ctx.fillRect(x-12+i*12,j*14,12,14);
    }
    ctx.fillStyle='#ffee58';ctx.shadowColor='#ffee58';ctx.shadowBlur=10;
    ctx.beginPath();ctx.arc(x,90,5,0,Math.PI*2);ctx.fill();ctx.shadowBlur=0;
  }
}

const boardEl=document.getElementById('board');
const chessOverlay=document.getElementById('chessOverlay');
const chessFeedback=document.getElementById('chessFeedback');
const chessMsg=document.getElementById('chessMsg');
let activePuzzle=null;
let selectedSq=null;

function buildBoard(puzzle){
  activePuzzle=puzzle;selectedSq=null;
  chessFeedback.textContent='';boardEl.innerHTML='';
  for(let r=0;r<8;r++) for(let c=0;c<8;c++){
    const sq=document.createElement('div');
    const light=(r+c)%2===0;
    sq.style.cssText='display:flex;align-items:center;justify-content:center;font-size:42px;line-height:1;cursor:pointer;-webkit-font-smoothing:antialiased;font-family:"Segoe UI Symbol","Arial Unicode MS",sans-serif;background:'+(light?'#eeeed2':'#769656')+';';
    const piece=puzzle.fen[r][c];
    if(piece){
      sq.textContent=PIECE_GLYPH[piece];
      const isW=(piece===piece.toUpperCase());
      sq.style.color=isW?'#fafafa':'#1a1a1a';
      sq.style.textShadow=isW?'0 1px 2px rgba(0,0,0,0.7),0 0 1px #000':'0 1px 2px rgba(255,255,255,0.4)';
    }
    sq.dataset.r=r;sq.dataset.c=c;
    sq.addEventListener('click',()=>onSquare(r,c,sq));
    boardEl.appendChild(sq);
  }
}
function squareAt(r,c){return boardEl.children[r*8+c];}
function clearHi(){for(const sq of boardEl.children) sq.style.outline='';}
function redrawBoard(){
  for(let r=0;r<8;r++) for(let c=0;c<8;c++){
    const sq=squareAt(r,c);const piece=activePuzzle.fen[r][c];
    sq.textContent=piece?PIECE_GLYPH[piece]:'';
    if(piece){
      const isW=(piece===piece.toUpperCase());
      sq.style.color=isW?'#fafafa':'#1a1a1a';
      sq.style.textShadow=isW?'0 1px 2px rgba(0,0,0,0.7),0 0 1px #000':'0 1px 2px rgba(255,255,255,0.4)';
    }
  }
  clearHi();
}
function onSquare(r,c,sq){
  const piece=activePuzzle.fen[r][c];
  if(selectedSq===null){
    if(piece && piece===piece.toUpperCase()){
      selectedSq=[r,c];clearHi();
      sq.style.outline='3px solid #4caf50';sq.style.outlineOffset='-3px';
    }
  } else {
    const [fr,fc]=selectedSq;
    const [tr,tc]=activePuzzle.to;
    const [sfr,sfc]=activePuzzle.from;
    if(fr===sfr&&fc===sfc&&r===tr&&c===tc){
      chessFeedback.style.color='#4caf50';
      chessFeedback.textContent='Checkmate! '+activePuzzle.desc;
      activePuzzle.fen[r][c]=activePuzzle.fen[fr][fc];
      activePuzzle.fen[fr][fc]='';
      redrawBoard();
      setTimeout(()=>{
        chessOverlay.style.display='none';
        if(currentLevel%3===0) showTrivia(); else advanceLevel();
      },1400);
    } else {
      chessFeedback.style.color='#ef5350';
      chessFeedback.textContent='Not mate. Try again!';
      selectedSq=null;clearHi();
    }
  }
}
function showChessPuzzle(){
  const p=nextPuzzle();
  const copy={from:p.from.slice(),to:p.to.slice(),desc:p.desc,fen:p.fen.map(row=>row.slice())};
  buildBoard(copy);
  chessMsg.textContent='White to move. Find mate in one to unlock level '+(currentLevel+1)+'!';
  chessOverlay.style.display='flex';
}

function loop(){
  requestAnimationFrame(loop);
  if(!game) return;
  game.update();game.draw();
  if(game.state==='win'){
    game.state='shown';showChessPuzzle();
  } else if(game.state==='dead'){
    timerRunning=false;
    const el=Math.floor((Date.now()-timerStart)/1000);
    const m=String(Math.floor(el/60)).padStart(2,'0');
    const s=String(el%60).padStart(2,'0');
    oTitle.textContent='GAME OVER';
    oMsg.textContent='You reached level '+currentLevel+' with '+game.score+' points in '+m+':'+s+'. Give it another shot!';
    oBtn.textContent='TRY AGAIN';oEl.style.display='flex';game.state='shown';
  }
}
